  let stepNum = 0
  let subNum = null
  let prefix = ''

    prefix = match[1]

    subNum = match[3] ? parseInt(match[3], 10) : null

      if (suffix) {
        if (subNum !== null) {
          mappedId = `math_${stepNum}_${suffix}_${subNum}`
        } else {
          mappedId = `math_${stepNum}_${suffix}`
        }
      }
    } else if (prefix === 'excel' && stepNum <= 10) {
      const suffixes: Record<number, string> = {
        1: 'intro',

        4: 'lookup',
        5: 'cleaning',
        6: 'pivots',
        7: 'powerquery',
        8: 'charts',
        9: 'analysis',
        10: 'macros'
      }
      const suffix = suffixes[stepNum]
      if (suffix) {
        if (subNum !== null) {
          mappedId = `excel_${stepNum}_${suffix}_${subNum}`
        } else {
          mappedId = `excel_${stepNum}_${suffix}`
        }
      }
    } else if (prefix === 'sheets' && stepNum <= 10) {

        1: 'intro',

            if (canvas && Array.isArray(canvas.nodes)) {
              const node = canvas.nodes.find((n: any) => n.id === nodeId)
              if (node) {
                return { title: node.title, text: node.text }
              }
            }
          }
        }
      } catch (e) {
        console.error("Error finding node title", e)
      }
      return null
    })()
